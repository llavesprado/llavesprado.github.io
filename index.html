const { WebSocketServer } = require('ws'); const http = require('http'); const server = http.createServer((req, res) => { res.writeHead(200, { 'Content-Type': 'text/plain; charset=utf-8' }); res.end('Servidor puente de Orión activo hacia Ollama (Puerto 11434).\n'); }); const wss = new WebSocketServer({ server }); wss.on('connection', (ws) => { console.log('¡Conexión establecida con la interfaz vía Tailscale!'); ws.on('message', async (message) => { try { let promptText = message.toString(); try { const parsed = JSON.parse(message); if (parsed.prompt) promptText = parsed.prompt; } catch (e) {} console.log('Mensaje recibido para la IA:', promptText); const ollamaData = JSON.stringify({ model: "llama3", // Cambia por tu modelo instalado (ej. 'phi3' o 'mistral') prompt: promptText, stream: false }); const ollamaReq = http.request({ hostname: '127.0.0.1', port: 11434, path: '/api/generate', method: 'POST', headers: { 'Content-Type': 'application/json', 'Content-Length': Buffer.byteLength(ollamaData) } }, (ollamaRes) => { let responseBody = ''; ollamaRes.on('data', (chunk) => { responseBody += chunk; }); ollamaRes.on('end', () => { try { const result = JSON.parse(responseBody); const respuestaIA = result.response || (result.message && result.message.content) || "Orión procesó la solicitud pero no generó texto."; ws.send(JSON.stringify({ response: respuestaIA })); console.log('Respuesta enviada de vuelta al cliente:', respuestaIA); } catch (err) { console.error('Error al parsear Ollama:', err); ws.send(JSON.stringify({ response: "Error interpretando la respuesta de la IA." })); } }); }); ollamaReq.on('error', (e) => { console.error('Error conectando a Ollama:', e.message); ws.send(JSON.stringify({ response: "Error: No se pudo conectar con el motor neuronal en el puerto 11434." })); }); ollamaReq.write(ollamaData); ollamaReq.end(); } catch (e) { console.error('Error general en el manejador de mensajes:', e); } }); ws.on('close', () => { console.log('Conexión cerrada.'); }); }); server.listen(3000, '0.0.0.0', () => { console.log('Servidor maestro de Orión corriendo en http://0.0.0.0:3000'); });
